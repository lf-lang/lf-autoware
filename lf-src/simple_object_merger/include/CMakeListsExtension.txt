unset(BUILD_TESTING)
set(LF_AUTOWARE_HOME $ENV{LF_AUTOWARE_HOME})
message(STATUS "Using ${LF_AUTOWARE_HOME} as LF Autoware home directory")

# Fix tinyxml2_vendor issue: pre-set TinyXML2 vars so vendor creates correct target
set(TinyXML2_FOUND TRUE)
set(TINYXML2_LIBRARY "/usr/lib/x86_64-linux-gnu/libtinyxml2.so")
set(TINYXML2_INCLUDE_DIR "/usr/include")


include_directories(
    ${LF_AUTOWARE_HOME}/lf-src/lf-include
)

find_package(autoware_simple_object_merger REQUIRED)
find_package(autoware_perception_msgs REQUIRED)
find_package(autoware_utils REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)

ament_target_dependencies(${LF_MAIN_TARGET} PUBLIC
    autoware_simple_object_merger
    autoware_perception_msgs
    autoware_utils
    geometry_msgs
    rclcpp
    rclcpp_components
)

find_library(LIB_SIMPLE_OBJECT_MERGER autoware_simple_object_merger
    PATHS ${autoware_simple_object_merger_DIR}/../../../lib NO_DEFAULT_PATH)
if(LIB_SIMPLE_OBJECT_MERGER)
    target_link_libraries(${LF_MAIN_TARGET} PUBLIC ${LIB_SIMPLE_OBJECT_MERGER})
endif()
