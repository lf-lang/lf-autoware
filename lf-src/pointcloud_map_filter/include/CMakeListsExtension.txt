unset(BUILD_TESTING)
set(LF_AUTOWARE_HOME $ENV{LF_AUTOWARE_HOME})
message(STATUS "Using ${LF_AUTOWARE_HOME} as LF Autoware home directory")

# Fix tinyxml2_vendor issue: pre-set TinyXML2 vars so vendor creates correct target
set(TinyXML2_FOUND TRUE)
set(TINYXML2_LIBRARY "/usr/lib/x86_64-linux-gnu/libtinyxml2.so")
set(TINYXML2_INCLUDE_DIR "/usr/include")

include_directories(
    ${LF_AUTOWARE_HOME}/lf-src/lf-include
    ${LF_AUTOWARE_HOME}/src/universe/autoware_universe/perception/autoware_compare_map_segmentation/src/voxel_based_approximate_compare_map_filter
)

# Create include alias so "autoware_compare_map_segmentation/node.hpp" resolves
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/include_alias/autoware_compare_map_segmentation)
file(GLOB CMS_HEADERS "${LF_AUTOWARE_HOME}/src/universe/autoware_universe/perception/autoware_compare_map_segmentation/src/voxel_based_approximate_compare_map_filter/*.hpp")
foreach(hdr ${CMS_HEADERS})
    get_filename_component(hdr_name ${hdr} NAME)
    file(CREATE_LINK ${hdr} ${CMAKE_BINARY_DIR}/include_alias/autoware_compare_map_segmentation/${hdr_name} SYMBOLIC)
endforeach()
include_directories(${CMAKE_BINARY_DIR}/include_alias)

find_package(autoware_compare_map_segmentation REQUIRED)
find_package(autoware_pointcloud_preprocessor REQUIRED)
find_package(autoware_point_types REQUIRED)
find_package(autoware_map_msgs REQUIRED)
find_package(autoware_utils REQUIRED)
find_package(diagnostic_updater REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)

ament_target_dependencies(${LF_MAIN_TARGET} PUBLIC
    autoware_compare_map_segmentation
    autoware_pointcloud_preprocessor
    autoware_point_types
    autoware_map_msgs
    autoware_utils
    diagnostic_updater
    geometry_msgs
    pcl_conversions
    rclcpp
    rclcpp_components
    sensor_msgs
    tf2
    tf2_ros
)

# Fix transitive curl dependency issue from libgdal
find_package(CURL REQUIRED)
target_link_libraries(${LF_MAIN_TARGET} PUBLIC CURL::libcurl)

# Link node implementation library
find_library(LIB_COMPARE_MAP_SEGMENTATION autoware_compare_map_segmentation
    PATHS ${autoware_compare_map_segmentation_DIR}/../../../lib NO_DEFAULT_PATH)
if(LIB_COMPARE_MAP_SEGMENTATION)
    target_link_libraries(${LF_MAIN_TARGET} PUBLIC ${LIB_COMPARE_MAP_SEGMENTATION})
endif()
