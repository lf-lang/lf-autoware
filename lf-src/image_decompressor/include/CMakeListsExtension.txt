    ${LF_AUTOWARE_HOME}/lf-src/lf-include
)

    autoware_image_transport_decompressor
    cv_bridge
    rclcpp
    rclcpp_components
    sensor_msgs
)

    target_link_libraries(${LF_MAIN_TARGET} PUBLIC ${LIB_IMAGE_DECOMPRESSOR})
