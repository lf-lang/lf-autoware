unset(BUILD_TESTING)
set(LF_AUTOWARE_HOME $ENV{LF_AUTOWARE_HOME})
message(STATUS "Using ${LF_AUTOWARE_HOME} as LF Autoware home directory")

include_directories(
    ${LF_AUTOWARE_HOME}/lf-src/lf-include
    ${LF_AUTOWARE_HOME}/src/universe/autoware_universe/control/autoware_control_validator/include
)

find_package(autoware_control_validator REQUIRED)
find_package(autoware_control_msgs REQUIRED)
find_package(autoware_motion_utils REQUIRED)
find_package(autoware_planning_msgs REQUIRED)
find_package(autoware_signal_processing REQUIRED)
find_package(autoware_utils REQUIRED)
find_package(autoware_vehicle_info_utils REQUIRED)
find_package(diagnostic_updater REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(visualization_msgs REQUIRED)

ament_target_dependencies(${LF_MAIN_TARGET} PUBLIC
    autoware_control_validator
    autoware_control_msgs
    autoware_motion_utils
    autoware_planning_msgs
    autoware_signal_processing
    autoware_utils
    autoware_vehicle_info_utils
    diagnostic_updater
    geometry_msgs
    nav_msgs
    rclcpp
    rclcpp_components
    visualization_msgs
)

# Link node implementation libraries
find_library(LIB_CONTROL_VALIDATOR autoware_control_validator_component
    PATHS ${autoware_control_validator_DIR}/../../../lib NO_DEFAULT_PATH)
find_library(LIB_CONTROL_VALIDATOR_HELPERS autoware_control_validator_helpers
    PATHS ${autoware_control_validator_DIR}/../../../lib NO_DEFAULT_PATH)
if(LIB_CONTROL_VALIDATOR)
    target_link_libraries(${LF_MAIN_TARGET} PUBLIC ${LIB_CONTROL_VALIDATOR})
endif()
if(LIB_CONTROL_VALIDATOR_HELPERS)
    target_link_libraries(${LF_MAIN_TARGET} PUBLIC ${LIB_CONTROL_VALIDATOR_HELPERS})
endif()
